COMMAND: sh -c 
    find /sys/class/backlight -maxdepth 2 -type f -print 2>/dev/null |
        sort |
        while IFS= read -r f; do
            case "$f" in
                */brightness|*/actual_brightness|*/max_brightness|*/bl_power|*/type|*/scale)
                    printf "%s=" "$f"
                    cat "$f"
                    ;;
            esac
        done



EXIT_CODE=0
